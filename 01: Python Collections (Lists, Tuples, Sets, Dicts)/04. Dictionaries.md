```python
# Like JS objects or Go maps, but more powerful
person = {
    "name": "Alice",
    "age": 25,
    "skills": ["Python", "Go", "React"]
}

# Accessing
print(person["name"])           # Alice
print(person.get("salary", 0))  # 0 (default if key missing, no error!)

# Adding/Updating
person["city"] = "NYC"
person["age"] = 26

# Safe removal
del person["city"]              # Removes key
popped = person.pop("age")      # Removes and returns value

# Iterating (3 ways)
for key in person:              # Just keys
    print(key)

for key, value in person.items():   # Key-value pairs
    print(f"{key}: {value}")

for value in person.values():       # Just values
    print(value)

# Dictionary comprehension
squares = {x: x**2 for x in range(5)}
# {0: 0, 1: 1, 2: 4, 3: 9, 4: 16}
```
