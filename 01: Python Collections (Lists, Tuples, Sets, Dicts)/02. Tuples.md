```python
# Tuples are immutable - use when data shouldn't change
point = (3, 4)
rgb = (255, 128, 0)

# Unpacking (very Pythonic)
x, y = point          # x=3, y=4
r, g, b = rgb         # r=255, g=128, b=0

# Swapping values (no temp variable needed!)
a, b = 5, 10
a, b = b, a           # a=10, b=5

# Tuple with one element needs comma
single = (5,)         # NOT (5) - that would just be int 5
```
